/**
 * Every value Mallan's amenity map binds to must be a member the provider actually publishes for that
 * field. A binding to a non-member is a DEAD filter: it can never match a listing.
 *
 * Found mechanically 2026-09-08 (18 of 73 bindings dead): ExteriorFeatures 'RoofDeck' (live:
 * BuildingRoofDeck, 2,487 Active rows), BuildingFeatures 'Elevator' (live: Elevators, 4,811),
 * PetsAllowed 'UnitYes' (live: Yes 2,837 / BuildingYes 6,396 — neither was in the map), View 'Park'
 * (live: ParkGreenbelt), InteriorFeatures 'Renovated'/'GutRenovated'/'NewlyRenovated'/'NaturalLight'/
 * 'Quiet'/'WoodBurningFireplace'/'DecorativeFireplace'/'WalkInCloset'/'HighCeiling', LaundryFeatures
 * 'OnCommonFloor', ListingTerms 'NoFee'/'OwnerPays'.
 *
 * Source of truth: data/cotality-contract/lookups.live.json (live Lookup catalogue, per resource+field).
 */
import { readFileSync } from 'node:fs';
import path from 'node:path';

import { AMENITY_FIELD_MAP } from '@/lib/search/types';

const ROOT = path.resolve(__dirname, '../..');
const lookups = JSON.parse(readFileSync(path.join(ROOT, 'data/cotality-contract/lookups.live.json'), 'utf8')) as Record<string, Record<string, { members: string[] }>>;
const compact = JSON.parse(readFileSync(path.join(ROOT, 'data/cotality-contract/contract.compact.json'), 'utf8')) as { resources: { Property: { fields: Record<string, unknown> } } };

describe('AMENITY_FIELD_MAP binds only live Property fields and live Lookup members', () => {
  const dead: string[] = [];
  for (const [key, mapping] of Object.entries(AMENITY_FIELD_MAP)) {
    for (const field of mapping.field.split(',').map((f) => f.trim())) {
      if (!compact.resources.Property.fields[field]) { dead.push(`${key}: field ${field} is not declared`); continue; }
      const members = lookups.Property[field]?.members;
      if (!members) continue; // a plain string field without a published vocabulary cannot be member-checked
      for (const v of mapping.values) if (!members.includes(v)) dead.push(`${key}: ${field} has no member "${v}"`);
    }
  }

  it('has no dead binding', () => {
    expect(dead).toEqual([]);
  });
});
